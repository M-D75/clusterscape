import{suppliedSlurmSummary,suppliedNotice}from'./slurm-summary';
import type{Snapshot}from'../../packages/contracts';import{canStart}from'../../packages/cluster-model';import{syntheticProfiles,cluster31Partitions,cluster31NodeOverrides}from'./cluster31-config';
export function cluster31(sequence:number,epoch:string):Snapshot{
 const tick=sequence%30;const base=Date.parse('2026-10-13T12:00:00.000Z')+Math.floor(sequence/30)*60000;const nodeId=(i:number)=>`node${String(i).padStart(2,'0')}`;
 const partitions=cluster31Partitions.map(([id,,,,limitHours])=>{const ref=suppliedSlurmSummary[id as keyof typeof suppliedSlurmSummary];const{hours,unlimited,...reference}=ref;return{id,name:id,limitHours:hours,limitUnlimited:unlimited,isDefault:id==='normal',reference};});
 const nodes:Snapshot['state']['nodes']=Array.from({length:31},(_,i)=>{const group=cluster31Partitions.find(([,start,end])=>i>=start&&i<=end)!;const ref=suppliedSlurmSummary[group[0] as keyof typeof suppliedSlurmSummary];const profile={...syntheticProfiles[group[3]],cpus:Math.max(syntheticProfiles[group[3]].cpus,ref.cpus),memory:Math.max(syntheticProfiles[group[3]].memory,Math.ceil(ref.memoryMiB/1024)),gpus:ref.gpus,...cluster31NodeOverrides[i]};return{id:nodeId(i),name:nodeId(i),kind:profile.gpus?'gpu':group[3]==='standard'?'standard':'memory',...profile,partitions:[group[0]],status:i===15&&tick>=14&&tick<20?'maintenance':'ready',usage:i===0&&tick>=10&&tick<15?null:0};});
 const users=Array.from({length:12},(_,i)=>`chercheur-${i+1}`);
 const targets=[0,15,1,19,23,26,28,30,0,15,3,16,20,24,27,29,4,30];
 const names=['Calcul normal','Calcul court','Calcul multi-nœuds','Calcul long','Mémoire partagée','Grande mémoire','Calcul global','Calcul GPU'];
 const jobs:Snapshot['state']['jobs']=targets.map((target,i)=>{const n=nodes[target];let status:Snapshot['state']['jobs'][number]['status']=i<8?'running':i<15?'queued':i===17?'failed':'finished';if(i===0&&tick>=6||i===1&&tick>=12)status='finished';if(i===8&&tick>=8)status=tick<12?'running':'finished';if(i===9&&tick>=20)status=tick<28?'running':'finished';const request={cpus:8,memory:32,gpus:n.gpus?1:0,nodeCount:i===2?2:1};const allocations=status==='running'?[{nodeId:n.id,cpus:8,memory:32,gpus:request.gpus}]:[];if(i===2&&status==='running')allocations.push({nodeId:nodes[2].id,cpus:8,memory:32,gpus:0});return{id:`t${i+1}`,name:names[i%names.length],submittedAt:new Date(base).toISOString(),user:users[i%users.length],partition:n.partitions[0],status,request,allocations,waitingReason:status==='queued'?(i===8?'Créneau réservé pas encore ouvert':i===9?tick>=14?'Nœud cible en maintenance':'Priorité du scénario, démarrage non garanti':'Ressources ou priorité du scheduler'):null};});
 const reservations:Snapshot['state']['reservations']=[{id:`r-${Math.floor(sequence/30)}`,jobId:'t9',startsAt:new Date(base+16000).toISOString(),endsAt:new Date(base+24000).toISOString(),status:tick<8?'planned':tick<12?'active':'finished',allocations:[{nodeId:nodeId(0),cpus:8,memory:32,gpus:0}]}];
 const s:Snapshot={version:2,cluster:{id:'cluster31',name:'Cluster 31 · hypothèse simulée'},stream:{epoch},sequence,observedAt:new Date().toISOString(),clock:{mode:'simulated',now:new Date(base+tick*2000).toISOString(),rate:1},units:{cpus:'logical-cpu',memory:'GiB',gpus:'device',usage:'percent-cpu-capacity'},topology:{partitions,users,referenceNotice:suppliedNotice,communication:{id:'communication-centre',name:'Centre de communication',description:'Repère logique de communication simulée ; aucun serveur physique décrit.'}},state:{nodes,jobs,reservations}};
 for(const i of [8,9]){const j=jobs[i];if(j.status==='running'&&!canStart(s,j.id,j.allocations)){j.status='queued';j.allocations=[];j.waitingReason='Contraintes de capacité ou disponibilité non satisfaites';}}
 for(let i=0;i<nodes.length;i++){const n=nodes[i];if(n.usage!==null)n.usage=jobs.some(j=>j.allocations.some(a=>a.nodeId===n.id))?45+(sequence*7+i*11)%45:0;}
 for(let i=0;i<nodes.length;i++){const n=nodes[i];const allocated=jobs.flatMap(j=>j.allocations).filter(a=>a.nodeId===n.id).reduce((v,a)=>v+a.memory,0);n.memoryUsed=i===0&&tick>=10&&tick<15?null:Math.min(n.memory,Math.round(n.memory*.06+allocated*(.45+((sequence+i)%7)/20)));}
 return s;
}
